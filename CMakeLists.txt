cmake_minimum_required(VERSION 3.27)
project(hyprune_core VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -fno-gnu-unique)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland>=0.56 hyprland<0.57)
pkg_check_modules(PLATFORM REQUIRED IMPORTED_TARGET pixman-1 libdrm xkbcommon hyprgraphics pangocairo wayland-server lua)
pkg_check_modules(SPATIAL REQUIRED IMPORTED_TARGET libxml-2.0 openssl)
add_library(hyprune_runtime STATIC src/interaction/Hands.cpp src/interaction/Config.cpp src/world/Environment.cpp src/world/Mounts.cpp src/surfaces/Layout.cpp src/world/Spatial.cpp src/control/Control.cpp src/world/World.cpp src/world/Schema.cpp src/sim/Camera.cpp src/surfaces/Surface.cpp src/ipc/Server.cpp src/ipc/Bridge.cpp)
target_link_libraries(hyprune_runtime PUBLIC PkgConfig::SPATIAL)
target_include_directories(hyprune_runtime PUBLIC src)
target_include_directories(hyprune_runtime SYSTEM PUBLIC third_party)
add_library(hyprune-core SHARED src/platform/hyprland/Plugin.cpp src/platform/hyprland/Capture.cpp src/render/Renderer.cpp)
set_target_properties(hyprune-core PROPERTIES PREFIX "")
target_link_libraries(hyprune-core PRIVATE hyprune_runtime PkgConfig::HYPRLAND PkgConfig::PLATFORM GLESv2 EGL)
add_executable(hypruned src/daemon/Main.cpp)
target_link_libraries(hypruned PRIVATE hyprune_runtime)
target_compile_definitions(hyprune-core PRIVATE HYPRUNE_DAEMON_PATH="${CMAKE_CURRENT_BINARY_DIR}/hypruned")
include(CTest)
if(BUILD_TESTING)
  pkg_check_modules(WAYLAND_CLIENT REQUIRED IMPORTED_TARGET wayland-client)
  find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/virtual-pointer.h ${CMAKE_CURRENT_BINARY_DIR}/virtual-pointer.c
    COMMAND ${WAYLAND_SCANNER} client-header ${CMAKE_CURRENT_SOURCE_DIR}/dev/nested/virtual-pointer.xml ${CMAKE_CURRENT_BINARY_DIR}/virtual-pointer.h
    COMMAND ${WAYLAND_SCANNER} private-code ${CMAKE_CURRENT_SOURCE_DIR}/dev/nested/virtual-pointer.xml ${CMAKE_CURRENT_BINARY_DIR}/virtual-pointer.c)
  add_executable(nested-pointer dev/nested/pointer.c ${CMAKE_CURRENT_BINARY_DIR}/virtual-pointer.c)
  target_include_directories(nested-pointer PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(nested-pointer PRIVATE PkgConfig::WAYLAND_CLIENT)
  add_executable(material-benchmark tests/material-benchmark.cpp src/render/Renderer.cpp)
  target_link_libraries(material-benchmark PRIVATE hyprune_runtime GLESv2 EGL)
  add_executable(render-tests tests/render.cpp src/render/Renderer.cpp)
  target_link_libraries(render-tests PRIVATE hyprune_runtime GLESv2 EGL)
  add_test(NAME render COMMAND render-tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/atrium)
  set_tests_properties(render PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1")
  add_executable(ipc-tests tests/ipc.cpp)
  target_link_libraries(ipc-tests PRIVATE hyprune_runtime)
  add_test(NAME ipc COMMAND ipc-tests)
  add_executable(control-tests tests/control.cpp)
  target_link_libraries(control-tests PRIVATE hyprune_runtime)
  add_test(NAME control COMMAND control-tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/atrium)
  add_executable(core-tests tests/core.cpp)
  target_link_libraries(core-tests PRIVATE hyprune_runtime)
  add_test(NAME core COMMAND core-tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/atrium)
endif()
if(BUILD_TESTING)
  add_executable(m2-tests tests/m2.cpp)
  target_link_libraries(m2-tests PRIVATE hyprune_runtime)
  add_test(NAME m2 COMMAND m2-tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/m2)
endif()
add_executable(hyprune-world-check src/daemon/WorldCheck.cpp)
target_link_libraries(hyprune-world-check PRIVATE hyprune_runtime)

if(BUILD_TESTING)
 add_executable(m3a-tests tests/m3a.cpp)
 target_link_libraries(m3a-tests PRIVATE hyprune_runtime)
 add_test(NAME m3a COMMAND m3a-tests)
endif()

if(BUILD_TESTING)
 pkg_check_modules(GTK3 IMPORTED_TARGET gtk+-3.0)
 if(GTK3_FOUND)
  add_executable(nested-parity-client dev/nested/parity-client.c)
  target_link_libraries(nested-parity-client PRIVATE PkgConfig::GTK3)
 endif()
endif()
