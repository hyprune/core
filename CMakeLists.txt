cmake_minimum_required(VERSION 3.27)
project(hyprune_core VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -fno-gnu-unique)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland>=0.56 hyprland<0.57)
pkg_check_modules(PLATFORM REQUIRED IMPORTED_TARGET pixman-1 libdrm xkbcommon hyprgraphics pangocairo wayland-server lua)
add_library(hyprune_runtime STATIC src/world/World.cpp src/sim/Camera.cpp src/surfaces/Surface.cpp src/ipc/Server.cpp)
target_include_directories(hyprune_runtime PUBLIC src)
target_include_directories(hyprune_runtime SYSTEM PUBLIC third_party)
add_library(hyprune-core SHARED src/platform/hyprland/Plugin.cpp src/platform/hyprland/Capture.cpp src/render/Renderer.cpp)
set_target_properties(hyprune-core PROPERTIES PREFIX "")
target_link_libraries(hyprune-core PRIVATE hyprune_runtime PkgConfig::HYPRLAND PkgConfig::PLATFORM GLESv2 EGL)
include(CTest)
if(BUILD_TESTING)
  add_executable(core-tests tests/core.cpp)
  target_link_libraries(core-tests PRIVATE hyprune_runtime)
  add_test(NAME core COMMAND core-tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/atrium)
endif()
